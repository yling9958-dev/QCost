cmake_minimum_required(VERSION 3.20)
project(QMFReproduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_compile_options(-O3 -DNDEBUG -march=native -D_GLIBCXX_USE_CXX11_ABI=0)
add_library(qmap_core STATIC src/geometry.cpp)
target_include_directories(qmap_core PUBLIC include)
add_library(qmap_baselines STATIC src/baselines/formal_baselines.cpp src/baselines/learnedbench_adapters.cpp src/baselines/rsmi_adapter.cpp)
target_include_directories(qmap_baselines PRIVATE "${QMF_BASELINES}/../learnedbench-patched" "${QMF_BASELINES}/morton-nd/include")
target_compile_definitions(qmap_baselines PRIVATE QMAP_RSMI_AVAILABLE=0)
target_link_libraries(qmap_baselines PUBLIC qmap_core)
add_library(qmap_runtime SHARED src/qmap_nai/static_qmap_index.cpp src/runtime/runtime_c_api.cpp)
target_link_libraries(qmap_runtime PUBLIC qmap_baselines)
set_target_properties(qmap_runtime PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
foreach(pair "build_index" "static_query" "persistence_audit")
  add_executable(qmap_${pair} src/benchmark/${pair}.cpp)
  target_link_libraries(qmap_${pair} PRIVATE qmap_runtime)
  set_target_properties(qmap_${pair} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
add_executable(qmap_profile src/benchmark/qmf_profile.cpp)
target_link_libraries(qmap_profile PRIVATE qmap_runtime)
set_target_properties(qmap_profile PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_executable(qmap_postgres_pilot src/benchmark/postgres_pilot.cpp)
target_link_libraries(qmap_postgres_pilot PRIVATE PostgreSQL::PostgreSQL Threads::Threads)
set_target_properties(qmap_postgres_pilot PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
